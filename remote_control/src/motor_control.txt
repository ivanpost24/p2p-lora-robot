#include <Arduino.h>
#include "config.h"

// ============================================================
// main.cpp — COMBINED JOYSTICK + MOTOR TEST BUILD
//
// What this does:
//   1. Startup motor sequence — pulses each side individually
//      so you can confirm wiring before touching the joystick.
//   2. Main loop — joystick → tank mix → motors.
//   3. Button = e-stop (STBY LOW + motors coasted).
//   4. Full debug print every loop to Serial.
//   5. OLED shows live joystick and motor values.
//
// SCHEMATIC NOTES:
//   U1 = RIGHT side (FR, BR). U2 = LEFT side (FL, BL).
//   If left/right feel swapped, set SWAP_SIDES = true below.
//   If fwd/rev feel backwards, set INVERT_Y = true below.
//
// STBY NOTE:
//   Physical SW1 switch must be ON for motors to move.
//   GPIO (PIN_STBY) can still pull STBY LOW to e-stop
//   even with SW1 ON — but GPIO cannot override SW1 OFF.
// ============================================================

// ------------------------------------------------------------
// Polarity correction — flip these if direction is wrong
// after running the startup test sequence.
// ------------------------------------------------------------
#define INVERT_Y    false   // true if forward drives robot backward
#define INVERT_X    false   // true if left turn goes right

// ------------------------------------------------------------
// Tank drive mixer
//
//   left  = Y + X
//   right = Y - X
//
// Y and X are each in range -512...+512 from joystick module.
// Worst case sum = 1024, constrain clips to ±255.
// ------------------------------------------------------------
static void tank_mix(int y, int x, int &left_out, int &right_out) {
    left_out  = constrain(y + x, -255, 255);
    right_out = constrain(y - x, -255, 255);
}

// ------------------------------------------------------------
// drive() — applies SWAP_SIDES then calls motor functions
// ------------------------------------------------------------
static void drive(int left, int right) {
    motor_left(left);
    motor_right(right);
}

// ------------------------------------------------------------
// Serial debug print
//
// Prints a fixed-width line every loop so you can watch
// values scroll in the Serial Monitor. Format:
//
//   JOY raw(X:#### Y:####) proc(X:#### Y:####) BTN:----
//   MOTOR L:#### R:#### | ESTOP:N
//
// Paste the output into Serial Plotter to graph proc values.
// ------------------------------------------------------------
static void debug_print(int raw_x, int raw_y,
                        int proc_x, int proc_y,
                        bool btn, int left, int right,
                        bool estop) {
    Serial.printf(
        "JOY raw(X:%-5d Y:%-5d) proc(X:%-5d Y:%-5d) BTN:%-5s | "
        "MOTOR L:%-5d R:%-5d | ESTOP:%c\n",
        raw_x, raw_y,
        proc_x, proc_y,
        btn ? "PRESS" : "----",
        left, right,
        estop ? 'Y' : 'N'
    );
}

// ------------------------------------------------------------
// OLED debug layout
//
//   Row 0: "JOY TEST + MOTOR"
//   Row 1: blank
//   Row 2: Raw  X: ####
//   Row 3: Raw  Y: ####
//   Row 4: blank
//   Row 5: L motor: ####
//   Row 6: R motor: ####
//   Row 7: Btn/ESTOP status
// ------------------------------------------------------------
static void debug_oled(int raw_x, int raw_y,
                       int left, int right,
                       bool estop) {
    display_clear();
    display_print_line(0, "JOY+MOTOR TEST", 0);
    display_print_line(2, "Raw X:  ", raw_x);
    display_print_line(3, "Raw Y:  ", raw_y);
    display_print_line(5, "L mtr:  ", left);
    display_print_line(6, "R mtr:  ", right);
    display_print_line(7, "ESTOP:  ", estop ? 1 : 0);
    display_show();
}

// ------------------------------------------------------------
// Startup motor test sequence
//
// Pulses each side forward then backward one at a time.
// Watch which physical wheels spin to confirm wiring.
// Expected if wired correctly per schematic:
//   Step 1: LEFT  side wheels spin forward
//   Step 2: RIGHT side wheels spin forward
//   Step 3: LEFT  side wheels spin backward
//   Step 4: RIGHT side wheels spin backward
//
// If something unexpected spins, note it and flip SWAP_SIDES
// or INVERT_Y above. Motors are stopped between each step.
// ------------------------------------------------------------
static void startup_motor_test() {
    Serial.println("=== STARTUP MOTOR TEST ===");
    Serial.println("Watch which wheels spin at each step.");
    Serial.println("SW1 physical switch must be ON.");

    // Step 1: Left forward
    Serial.printf("Step 1 — LEFT  FORWARD  (duty %d)...\n", TEST_DUTY);
    motor_left(TEST_DUTY);
    motor_right(0);
    delay(TEST_DURATION_MS);
    motors_stop();
    delay(200);

    // Step 2: Right forward
    Serial.printf("Step 2 — RIGHT FORWARD  (duty %d)...\n", TEST_DUTY);
    motor_left(0);
    motor_right(TEST_DUTY);
    delay(TEST_DURATION_MS);
    motors_stop();
    delay(200);

    // Step 3: Left backward
    Serial.printf("Step 3 — LEFT  BACKWARD (duty %d)...\n", TEST_DUTY);
    motor_left(-TEST_DUTY);
    motor_right(0);
    delay(TEST_DURATION_MS);
    motors_stop();
    delay(200);

    // Step 4: Right backward
    Serial.printf("Step 4 — RIGHT BACKWARD (duty %d)...\n", TEST_DUTY);
    motor_left(0);
    motor_right(-TEST_DUTY);
    delay(TEST_DURATION_MS);
    motors_stop();
    delay(200);

    Serial.println("=== TEST COMPLETE — entering joystick control ===");
    Serial.println("Button = e-stop. Release to re-enable.");
    Serial.println();
}

// ============================================================
// Setup
// ============================================================
void setup() {
    Serial.begin(115200);
    // Small delay so CDC USB enumerates before first prints
    delay(500);

    Serial.println("=== Tank Drive Test Build ===");
    Serial.println("Initializing...");

    display_init();
    joystick_init();
    motors_init();     // sets up LEDC, pinMode, asserts STBY HIGH

    Serial.println("Init complete.");
    Serial.println("REMINDER: SW1 physical switch must be ON for motors.");
    Serial.println();

    // Run startup test so you can verify each motor before
    // joystick control begins. Remove call when wiring confirmed.
    // startup_motor_test();
}

// ============================================================
// Loop
// ============================================================

// int left =  -255;
// int right = -255;
void loop() {
    // // --- Read joystick -----------------------------------------
    int raw_x  = analogRead(PIN_VRX);
    int raw_y  = analogRead(PIN_VRY);
    bool btn   = (digitalRead(PIN_SW) == LOW);

    // Apply polarity flags before passing to mixer
    int proc_x = (INVERT_X ? -1 : 1) * (raw_x - ADC_CENTER);
    int proc_y = (INVERT_Y ? -1 : 1) * (raw_y - ADC_CENTER);

    // Deadzone
    if (proc_x > -ADC_DEADZONE && proc_x < ADC_DEADZONE) proc_x = 0;
    if (proc_y > -ADC_DEADZONE && proc_y < ADC_DEADZONE) proc_y = 0;

    // Scale to ±512 for mixer headroom
    proc_x /= 4;
    proc_y /= 4;

    // --- Mix ---------------------------------------------------
    int left, right;
    tank_mix(proc_y, proc_x, left, right);

    // --- Drive -------------------------------------------------
    drive(left, right);

    // --- Debug output ------------------------------------------
    //debug_print(raw_x, raw_y, proc_x, proc_y, btn, left, right, false);
    debug_oled(raw_x, raw_y, left, right, false);

    delay(LOOP_PERIOD_MS);
}
